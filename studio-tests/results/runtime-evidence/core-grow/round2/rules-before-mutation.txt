// 纯规则不依赖浏览器，算术顺序与原型一致。
export function advanceTime(elapsed: number, dt: number, duration: number): number {
  return Math.min(duration, elapsed + dt);
}
export function isCatch(marker: number, target: number, width: number): boolean {
  return Math.abs(marker - target) <= width * 0.5;
}
export function hasWon(score: number, quota: number): boolean {
  return score >= quota;
}
export function markerAt(phase: number): number {
  const p = phase % 2;
  return p <= 1 ? p : 2 - p;
}
export function makeRng(seed: number): () => number {
  return function rng(): number { // mulberry32; every gameplay random call reads this, never Math.random
  seed = (seed + 0x6D2B79F5) | 0;
  let t = Math.imul(seed ^ (seed >>> 15), seed | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};
}
